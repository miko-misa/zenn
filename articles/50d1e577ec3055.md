### 直和

```python
from math import tanh

import plotly.graph_objects as go
from plotly.subplots import make_subplots


def f(x, a):
  return tanh(x) + 0.2 * x + 0.3 - a * (x ** 2)


def df(x, a):
  return (1 - tanh(x) ** 2) + 0.2 - 2 * a * x


def newton(x, a, tol=1e-6, max_iter=1000, deceleration=False):
  guesses = []
  f_values = []
  for _ in range(max_iter):
    f_x = f(x, a)
    df_x = df(x, a)
    guesses.append(x)
    f_values.append(f_x)
    if abs(df_x) < 1e-12:
      break
    if deceleration:
      mu = 1.0
      while True:
        x_new = x - mu * f_x / df_x
        if abs(f(x_new, a)) < (1 - mu / 4) * abs(f_x):
          break
        mu *= 0.5
    else:
      x_new = x - f_x / df_x
    if abs(x_new - x) < tol:
      guesses.append(x_new)
      f_values.append(f(x_new, a))
      return x_new, guesses, f_values
    x = x_new
    print(f"Current guess: {x}, f(x): {f_values[-1]}")
  return x, guesses, f_values


def bisection(low, high, a, tol=1e-6, max_iter=1000):
  guesses = []
  f_values = []
  if f(low, a) * f(high, a) >= 0:
    raise ValueError("f(low) and f(high) must have opposite signs.")
  for _ in range(max_iter):
    mid = (low + high) / 2
    f_mid = f(mid, a)
    guesses.append(mid)
    f_values.append(f_mid)
    print(f"Current guess: {mid}, f(x): {f_mid}")
    if abs(f_mid) < tol:
      return mid, guesses, f_values
    if f(low, a) * f_mid < 0:
      high = mid
    else:
      low = mid
  return (low + high) / 2, guesses, f_values


def print_convergence(guesses, f_values):
  print("Iteration | Guess Value (x) | f(x) Value")
  for i, (guess, f_val) in enumerate(zip(guesses, f_values), 1):
    print(f"[{i - 1}], [{guess:.12f}], [{f_val:.6e}],")


METHOD = "newton"  # "newton" or "bisection"

if __name__ == "__main__":
  a = 1
  initial_guess = 1.2
  if METHOD == "newton":
    root, guesses, f_values = newton(initial_guess, a, deceleration=False)
  else:
    root, guesses, f_values = bisection(1, 10, a)
  print("------ Convergence Details ------")
  print_convergence(guesses, f_values)
  iterations = list(range(1, len(guesses) + 1))
  fig = make_subplots(specs=[[{"secondary_y": True}]])

  fig.add_trace(
    go.Scatter(
      x=iterations,
      y=guesses,
      mode="markers+lines",
      name="Guess value (x)",
    ),
    secondary_y=False,
  )

  fig.add_trace(
    go.Scatter(
      x=iterations,
      y=f_values,
      mode="markers+lines",
      name="f(x) value",
    ),
    secondary_y=True,
  )

  fig.update_layout(
    title="Convergence of " + METHOD + " Method",
    xaxis_title="Iteration (Step)",
  )

  fig.update_yaxes(title_text="Guess Value (x)", secondary_y=False)
  fig.update_yaxes(title_text="f(x) Value", secondary_y=True)
  fig.write_image("outputs/" + METHOD + "_convergence.svg")
  print(f"Root found: {root}, f(root): {f(root, a)}")
```
